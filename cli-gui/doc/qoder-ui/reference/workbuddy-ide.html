import { useState, useMemo } from "react";
import { useAgent } from "@workbuddy/agent";
 
// Agent Panel — 结对编程，带检查点回滚
export function AgentPanel() {
  const [mode, setMode] = useState<Mode>("agent");
  const { plan, run, checkpoint } = useAgent();
 
  const steps = useMemo(() => plan.steps, [plan]);
 
  return (
    <Panel mode={mode} onRun={() => run(steps)}>
      {steps.map((s) => (
        <Step key={s.id} done={s.done} />
      ))}
      {checkpoint && <Diff data={checkpoint.diff} />}
    </Panel>
  );
}
